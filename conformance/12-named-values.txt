# Named values. `let` sets, replaces and removes one. `{name}` stays in the stored text: the renderer
# substitutes when it draws. The board lists the values by name.
agent: let team 12 people
  -> ok
agent: let budget 1,200 USD
  -> ok
agent: let alpha spaces   inside   stay
  -> ok
agent: let team 14 people
  -> ok
agent: let gone soon
  -> ok
agent: let gone
  -> ok
agent: let never
  -> ok
agent: let 예산 30만원
  -> ok
agent: let due_2026-12 Friday
  -> ok
agent: let <24 n> longest name
  -> ok
agent: let <25 n> too long a name
  -> err bad-id
agent: let a!b 1
  -> err bad-id
agent: let {team} 1
  -> err bad-id
agent: let
  -> err missing-argument
agent: let exact <60 v>
  -> ok
agent: let long <61 v>
  -> ok clipped let.long 60
